import {spawn,execFileSync} from 'node:child_process';
import {mkdirSync,writeFileSync,readFileSync,openSync} from 'node:fs';
import {createHash} from 'node:crypto';
const root='D:/serrian-tide-world',out=root+'/.local/cube-d1-final-startup';mkdirSync(out+'/docs/evidence',{recursive:true});
if(!JSON.parse(readFileSync(root+'/.local/cube-d1-final-browser/validation-complete.json')).frozenSourcesUnchanged)throw Error('Full pre-correction regression not complete');
const initial=JSON.parse(readFileSync(root+'/.local/cube-d1-final-browser/source-freeze.json')).files;
const manifest=()=>Object.fromEntries(Object.keys(initial).map(p=>[p,createHash('sha256').update(readFileSync(root+'/'+p)).digest('hex')]));

try{
 await run('typecheck',['node_modules/typescript/bin/tsc','--noEmit']);await run('build',['node_modules/vite/bin/vite.js','build']);

 if(JSON.stringify(manifest())!==JSON.stringify(before))throw Error('Final sources changed during affected browser checks');

 if(JSON.stringify(manifest())!==JSON.stringify(before))throw Error('Final sources changed during performance');
 writeFileSync(out+'/validation-complete.json',JSON.stringify({status:'passed',at:new Date().toISOString(),frozenSourcesUnchanged:true,changedSinceFullRegression:changed,affectedBrowserCases:24,numericalAndLegacyDependenciesUnchanged:true}));
}catch(e){writeFileSync(out+'/validation-failed.json',JSON.stringify({error:String(e),at:new Date().toISOString()}));process.exitCode=1;}
