import { readFileSync, writeFileSync } from 'node:fs';
import assert from 'node:assert/strict';
const read=label=>JSON.parse(readFileSync(`docs/evidence/a7-p1c-${label}-measurements.json`,'utf8'));
const before=read('before'),after=read('after');
const quantile=values=>{const a=values.filter(Number.isFinite).sort((x,y)=>x-y);return{count:a.length,p50:a.length?a[Math.floor((a.length-1)*.5)]:null,p95:a.length?a[Math.ceil((a.length-1)*.95)]:null,max:a.at(-1)??null};};
const frames=r=>r.frames.slice(1).map((f,i)=>f.raf-r.frames[i].raf);
const key=r=>[r.backend,r.profile,r.n,r.mode,r.cost,r.tool,r.repeat].join('/');
const pairs=new Map(before.records.map(r=>[key(r),r]));assert.equal(before.records.length,384);assert.equal(after.records.length,384);
for(const r of after.records){const b=pairs.get(key(r));assert.ok(b);assert.equal(r.fixtureSha,b.fixtureSha);assert.equal(r.finalHash,b.finalHash,'Identical actual screen input must produce exactly the same final geography');}
const groups=[];
for(const backend of['hardware','fallback'])for(const profile of['desktop','mobile'])for(const n of[128,256])for(const mode of['globe','net'])for(const cost of['ordinary','large']){
 const pick=source=>source.records.filter(r=>r.backend===backend&&r.profile===profile&&r.n===n&&r.mode===mode&&r.cost===cost);
 const summarize=records=>({strokes:records.length,firstVisibleMs:quantile(records.map(r=>r.firstVisibleMs).filter(v=>v!==null)),noObservedChangedPixels:records.filter(r=>r.firstVisibleMs===null).length,paintFrameIntervalsMs:quantile(records.flatMap(r=>frames(r.result))),mainTasksMs:quantile(records.flatMap(r=>r.result.mainTasksMs)),beginMs:quantile(records.flatMap(r=>r.result.stages.begin)),pumpMs:quantile(records.flatMap(r=>r.result.stages.pump)),displayFieldMs:quantile(records.flatMap(r=>r.result.stages.updateDisplay)),queuePeak:Math.max(...records.flatMap(r=>r.result.frames.map(f=>f.pending))),worker:records.filter(r=>r.result.final.work?.backend==='worker').map(r=>({tool:r.tool,repeat:r.repeat,readyRoundTripMs:r.result.final.work.readyRoundTripMs,snapshotMs:r.result.final.work.snapshotMs,computeMs:r.result.final.work.computeMs,validationMaxMs:r.result.final.work.validationMaxMs,commitMaxMs:r.result.final.work.commitMaxMs,ackWait:r.result.final.work.ackWait,statistics:r.result.final.work.statistics}))});
 groups.push({backend,profile,n,mode,cost,before:summarize(pick(before)),after:summarize(pick(after))});
 const group=groups.at(-1);for(const[phase,source]of[['before',before],['after',after]]){const records=pick(source);group[phase].perStrokeMainTaskMaxMs=quantile(records.map(r=>Math.max(...r.result.mainTasksMs)));group[phase].longTasksMs=quantile(records.flatMap(r=>r.result.longTasks));}
}
const navigation=[];for(const b of before.navigation){const a=after.navigation.find(a=>a.backend===b.backend&&a.profile===b.profile&&a.n===b.n&&a.mode===b.mode);assert.ok(a);navigation.push({backend:b.backend,profile:b.profile,n:b.n,mode:b.mode,before:quantile(frames(b.result)),after:quantile(frames(a.result)),afterHeap:a.heap,afterRenderer:a.result.final.renderer,afterDiagnostics:a.result.final.diagnostics});}
const worker=after.records.filter(r=>r.result.final.work?.backend==='worker').map(r=>r.result.final.work);
const summary={base:before.base,method:before.method,pairs:384,fixtureAndFinalWordsExact:true,groups,navigation,worker:{tasks:worker.length,readyRoundTripMs:quantile(worker.map(w=>w.readyRoundTripMs)),snapshotMs:quantile(worker.map(w=>w.snapshotMs)),computeMs:quantile(worker.map(w=>w.computeMs)),validationMaxMs:quantile(worker.map(w=>w.validationMaxMs)),commitMaxMs:quantile(worker.map(w=>w.commitMaxMs)),acknowledgementWindowMs:quantile(worker.flatMap(w=>w.ackWait.samples)),acknowledgementLifetimeCount:worker.reduce((s,w)=>s+w.ackWait.count,0),acknowledgementLifetimeMaxMs:Math.max(...worker.map(w=>w.ackWait.maxMs))},scope:'All quantiles are raw measurement populations except acknowledgement windows, retained up to 512 per task. Page CPU throttling does not throttle dedicated workers or desktop GPU. PNG observation includes capture overhead. No physical device claim. Before hardware profile still uses the old CPU sphere renderer.'};
writeFileSync('docs/evidence/a7-p1c-performance-summary.json',JSON.stringify(summary,null,2));
const fmt=q=>[q.p50,q.p95,q.max].map(v=>v===null?'n/a':v.toFixed(1)).join(' / ');
let md='# A7-P1C matched spherical measurements\n\n384 matched held strokes, 16 navigation contexts per revision; exact saved-fixture and final-word hashes verified for every pair. Values are p50 / p95 / max in milliseconds. Desktop/mobile profiles run on one Windows host; mobile is Pixel 7 layout plus 4x page CPU throttling, not physical hardware or worker/GPU throttling. Baseline hardware profile uses the original CPU sphere renderer. Compositor PNG timing includes capture overhead.\n\n| Backend/profile/detail/view/cost | Visible before | Visible after | Main tasks before | Main tasks after |\n| --- | --- | --- | --- | --- |\n';for(const g of groups)md+=`| ${g.backend}/${g.profile}/N${g.n}/${g.mode}/${g.cost} | ${fmt(g.before.firstVisibleMs)} | ${fmt(g.after.firstVisibleMs)} | ${fmt(g.before.mainTasksMs)} | ${fmt(g.after.mainTasksMs)} |\n`;
md+='\nThe task columns above describe the entire task population. The table below instead takes each stroke\'s largest main-thread task, then reports that 12-stroke distribution. Long-task observations and both populations remain in JSON.\n\n| Backend/profile/detail/view/cost | Per-stroke worst task before | Per-stroke worst task after |\n| --- | --- | --- |\n';for(const g of groups)md+=`| ${g.backend}/${g.profile}/N${g.n}/${g.mode}/${g.cost} | ${fmt(g.before.perStrokeMainTaskMaxMs)} | ${fmt(g.after.perStrokeMainTaskMaxMs)} |\n`;
md+='\nNavigation intervals below measure RAF scheduling. They do not establish that a new coherent CPU image was presented on every RAF; fallback reconstruction can span multiple frames. Display diagnostics and completed-image timings remain separate.\n\n| Navigation backend/profile/detail/view | RAF interval before | RAF interval after |\n| --- | --- | --- |\n';for(const n of navigation)md+=`| ${n.backend}/${n.profile}/N${n.n}/${n.mode} | ${fmt(n.before)} | ${fmt(n.after)} |\n`;
md+=`\nWorker tasks: ${worker.length}. Ready round trip ${fmt(summary.worker.readyRoundTripMs)} ms; private snapshot ${fmt(summary.worker.snapshotMs)} ms; worker compute ${fmt(summary.worker.computeMs)} ms; per-task host validation maximum ${fmt(summary.worker.validationMaxMs)} ms; commit maximum ${fmt(summary.worker.commitMaxMs)} ms. Acknowledgement retained-window distribution ${fmt(summary.worker.acknowledgementWindowMs)} ms, lifetime count ${summary.worker.acknowledgementLifetimeCount}. These are distinct events, not end-to-end visible latency. Ready time includes private topology/snapshot setup on a preloaded worker.\n\nFull per-tool raw populations, queue peaks, actual graphics diagnostics and JS-realm heap readings are in [the machine-readable summary](a7-p1c-performance-summary.json). Heap readings include development/test overhead and do not represent total process/GPU/worker memory. See the completion report for interpretation and limitations.\n`;
writeFileSync('docs/evidence/a7-p1c-performance.md',md);
console.log({pairs:384,groups:groups.length,worker:worker.length,exactFinalHashes:true});
