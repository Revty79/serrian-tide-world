import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync, copyFileSync, createWriteStream, readdirSync } from 'node:fs';
const prefix = 'docs/evidence/cube-a-', sha = (buffer) => createHash('sha256').update(buffer).digest('hex');
const phases = JSON.parse(readFileSync(prefix + 'validation.json', 'utf8'));
assert.equal(phases.find((p) => p.label === 'browser-regression')?.exitCode, 0);
const browserReport = JSON.parse(readFileSync(prefix + 'browser-result.json', 'utf8'));
assert.equal(browserReport.stats.unexpected, 0); assert.equal(browserReport.stats.flaky, 0);
const browserSources = JSON.parse(readFileSync(prefix + 'tested-source.json', 'utf8'));
const changed = browserSources.files.filter((f) => sha(readFileSync(f.path)) !== f.sha256);
assert.deepEqual(changed.map((f) => f.path).sort(), ['src/surfaces/solid-cube.test.ts', 'src/surfaces/solid-cube.ts']);
copyFileSync(prefix + 'tested-source.json', prefix + 'browser-tested-source.json');
copyFileSync(prefix + 'source-freeze.json', prefix + 'browser-source-freeze.json');
const finalFiles = browserSources.files.map((f) => ({ path: f.path, sha256: sha(readFileSync(f.path)) }));
const registry = { baseline: browserSources.baseline, scope: 'Final numerical/type/build verification. Browser inputs unchanged; see source-bridge.', files: finalFiles };
writeFileSync(prefix + 'tested-source.json', JSON.stringify(registry, null, 2) + '\n');
const bridge = { reason: 'Post-browser sparse/array-like vector validation hardening in the unimported numerical-only cube provider; existing product and browser test sources unchanged.',
  browserImportedSourceChanges: 0, changedUnimportedCubeFiles: changed.map((f) => ({ path: f.path, browserRunSha256: f.sha256, finalSha256: sha(readFileSync(f.path)) })),
  unchangedRegistryEntries: browserSources.files.length - changed.length, browserResult: browserReport.stats, finalNumericalRegistry: prefix + 'tested-source.json' };
writeFileSync(prefix + 'source-bridge.json', JSON.stringify(bridge, null, 2) + '\n');
async function run(label, args) {
  const log = createWriteStream(prefix + label + '.txt'), started = Date.now(); console.log('start', label);
  const child = spawn(process.execPath, args, { windowsHide: true, stdio: ['ignore', 'pipe', 'pipe'] });
  child.stdout.pipe(log); child.stderr.pipe(log);
  const exitCode = await new Promise((resolve, reject) => { child.once('error', reject); child.once('exit', resolve); });
  await new Promise((resolve) => log.end(resolve));
  const record = { label, command: ['node', ...args], exitCode, elapsedMs: Date.now() - started };
  phases.push(record); writeFileSync(prefix + 'validation.json', JSON.stringify(phases, null, 2) + '\n'); console.log(record);
  assert.equal(exitCode, 0);
  for (const f of finalFiles) assert.equal(sha(readFileSync(f.path)), f.sha256, f.path);
}
await run('final-focused', ['--test', 'src/surfaces/solid-cube.test.ts', 'src/terrain/solid-cube.test.ts']);
await run('final-geometry-certificate', ['scripts/verify-solid-cube.mjs']);
const tests = ['src/terrain', 'src/rendering', 'src/editor', 'src/persistence', 'src/surfaces'].flatMap((d) => readdirSync(d).filter((f) => f.endsWith('.test.ts')).map((f) => `${d}/${f}`));
await run('final-numerical-regression', ['--test', ...tests]);
await run('final-typecheck', ['node_modules/typescript/bin/tsc', '--noEmit']);
await run('final-build', ['node_modules/vite/bin/vite.js', 'build']);
writeFileSync(prefix + 'source-freeze.json', JSON.stringify({ count: finalFiles.length, changed: [], digest: sha(JSON.stringify(finalFiles)),
  scope: 'Final focused/certificate/full numerical/typecheck/build; unchanged browser consumers proven by source-bridge and accepted product hashes.' }, null, 2) + '\n');
console.log('Final numerical checks complete; browser consumers unchanged.');
