const paths=[...new Set([...git('diff','--name-only',baseline,'--').split(/\r?\n/),...git('ls-files','--others','--exclude-standard').split(/\r?\n/)])].filter(Boolean).sort();
const files=paths.filter(p=>!p.endsWith('cube-d1-delivery-manifest.json')).map(path=>{const bytes=readFileSync(path);return{path,category:path.startsWith('docs/evidence/')?'evidence':path==='README.md'||path.startsWith('docs/')?'documentation':path.startsWith('tests/')||path.endsWith('.test.ts')?'test':'runtime-config',change:git('ls-tree',baseline,'--',path)?'modified':'added',bytes:bytes.length,sha256:hash(bytes)};});
if(files.some(f=>!f.path.startsWith('docs/evidence/cube-d1-')&&f.category==='evidence'))throw Error('Unrelated evidence changed');
writeFileSync('docs/evidence/cube-d1-delivery-manifest.json',JSON.stringify({baseline,generated:new Date().toISOString(),categories:Object.fromEntries(['runtime-config','test','documentation','evidence'].map(c=>[c,files.filter(f=>f.category===c).length])),note:'Working-byte SHA256 includes native line endings.
