import {readFileSync,writeFileSync,readdirSync,copyFileSync,existsSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {gunzipSync} from 'node:zlib';
import path from 'node:path';
const dir='docs/evidence',hash=b=>createHash('sha256').update(b).digest('hex');
for(const file of['report.mjs','inventory.mjs','reconcile.mjs','seal.mjs','pack-traces.py','apply-startup.mjs'])copyFileSync('.local/cube-d1-'+file,dir+'/cube-d1-'+file+'.txt');
const freeze=JSON.parse(readFileSync(dir+'/cube-d1-source-freeze.json'));for(const[p,sha]of Object.entries(freeze.files))if(hash(readFileSync(p))!==sha)throw Error('Final frozen source drift '+p);
const files=readdirSync(dir).filter(p=>p.startsWith('cube-d1-')&&!['cube-d1-delivery-manifest.json','cube-d1-evidence-manifest.json','cube-d1-implementation-files.json'].includes(p)).sort().map(file=>{const bytes=readFileSync(dir+'/'+file),entry={file,bytes:bytes.length,sha256:hash(bytes)};if(file.endsWith('.gz')){const raw=gunzipSync(bytes);entry.raw={name:file.slice(0,-3),bytes:raw.length,sha256:hash(raw)};}if(bytes.length>40*1024*1024)throw Error('Oversized review artifact '+file);return entry;});
const images=files.filter(f=>f.file.endsWith('.png'));if(images.length!==40)throw Error('Screenshot count');for(const f of images)if(hash(readFileSync('.local/cube-d1-final-startup/docs/evidence/'+f.file))!==f.sha256)throw Error('Screenshot copy mismatch');
const archives=JSON.parse(readFileSync(dir+'/cube-d1-trace-archives.json'));if(archives.archives.flatMap(a=>a.entries).length!==24)throw Error('Trace count');for(const archive of archives.archives)if(hash(readFileSync(dir+'/'+archive.file))!==archive.sha256)throw Error('Archive hash mismatch');
for(const doc of['docs/CUBE-D1-COMPLETION.md','docs/SOLID-CUBE-VISUAL-PREVIEW.md']){const s=readFileSync(doc,'utf8');if(/FINAL_[A-Z_]+|6,315,\?/.test(s))throw Error('Unfilled report');for(const m of s.matchAll(/\]\(([^)]+)\)/g)){const ref=m[1];if(/^(https?:|#)/.test(ref)||ref==='CUBE-D1-DELIVERY.md')continue;if(!existsSync(path.resolve(path.dirname(doc),ref.split('#')[0])))throw Error('Missing link '+doc+' '+ref);}}
writeFileSync(dir+'/cube-d1-evidence-manifest.json',JSON.stringify({generated:new Date().toISOString(),baseline:'77df76f3a0a12273f9277d201277f13e9985074d',files,sourceScope:'Only new D1 evidence; prior accepted evidence preserved. Current 414-case inventory is reconciled from the successful 412-case process and final 24-case affected run; it is not a single-process final 414-case claim.',compression:'Standard lossless gzip; raw and compressed SHA256/lengths included. ZIP trace parts contain exact original final traces, individually verified by SHA256/length. No image editing or trace repacking.'},null,2)+'\n');
console.log(JSON.stringify({evidenceFiles:files.length+1,screenshots:images.length,traces:24,sourceFiles:Object.keys(freeze.files).length,largestBytes:Math.max(...files.map(f=>f.bytes))}));
