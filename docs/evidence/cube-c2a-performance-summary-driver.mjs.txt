import {readFileSync,writeFileSync} from 'node:fs';
const prefix='docs/evidence/cube-c2a-',data=JSON.parse(readFileSync(prefix+'performance.json','utf8'));
const med=a=>{const x=[...a].sort((a,b)=>a-b);return x.length%2?x[(x.length-1)/2]:(x[x.length/2-1]+x[x.length/2])/2;};
const range=a=>({median:med(a),max:Math.max(...a)});
const pathGroups=[...new Set(data.trials.map(t=>t.label))].map(label=>{const a=data.trials.filter(t=>t.label===label);return {label,trials:a.length,statuses:a.map(t=>t.status),elapsedMs:range(a.map(t=>t.elapsedMs)),maxPumpMs:Math.max(...a.map(t=>t.maxPumpMs)),maxNodeYieldGapMs:Math.max(...a.map(t=>t.maxNodeYieldGapMs)),maxArcTerms:Math.max(...a.map(t=>t.statistics.peakArcTerms)),maxWork:Math.max(...a.map(t=>t.statistics.workUnits)),routePumpsMs:range(a.flatMap(t=>t.pumps.filter(p=>p.stage==='route').map(p=>p.ms))),centerPumpsMs:range(a.flatMap(t=>t.pumps.filter(p=>p.stage==='dab-center').map(p=>p.ms)))};});
const routeGroups=[...new Set(data.routeTrials.map(t=>t.label))].map(label=>({label,ms:range(data.routeTrials.filter(t=>t.label===label).map(t=>t.ms))}));
const dabGroups=[128,256].flatMap(n=>[40.96,563.2].map(radius=>{const a=data.dabTrials.filter(t=>t.n===n&&t.radiusMeters===radius);return {n,radius,trials:a.length,totalPreparationMs:range(a.map(t=>t.totalPreparationMs)),initializationMs:range(a.map(t=>t.initializationMs)),maxPumpMs:Math.max(...a.map(t=>t.maxPumpMs)),maxNodeYieldGapMs:Math.max(...a.map(t=>t.maxNodeYieldGapMs)),peakPreparedBytes:Math.max(...a.map(t=>t.statistics.peakPreparedBytes))};}));
const summary={pathGroups,routeGroups,dabGroups,expectedFailures:data.failures.map(t=>({label:t.label,status:t.status,retainedArcTerms:t.statistics.retainedArcTerms,workUnits:t.statistics.workUnits,failedRouteWorkAllowance:t.statistics.failedRouteWorkAllowance})),maxPathPumpMs:Math.max(...data.trials.map(t=>t.maxPumpMs)),maxDabPumpMs:Math.max(...data.dabTrials.map(t=>t.maxPumpMs))};
writeFileSync(prefix+'performance-summary.json',JSON.stringify(summary,null,2)+'\n');console.log(JSON.stringify(summary,null,2));
