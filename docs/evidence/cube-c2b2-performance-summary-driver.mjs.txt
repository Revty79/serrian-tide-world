import {readFileSync,writeFileSync,existsSync,renameSync} from 'node:fs';
import {gzipSync,gunzipSync} from 'node:zlib';
import {createHash} from 'node:crypto';
import assert from 'node:assert/strict';
const prefix='docs/evidence/cube-c2b2-',sha=b=>createHash('sha256').update(b).digest('hex');
const q=values=>{const a=values.filter(Number.isFinite).toSorted((a,b)=>a-b);return {count:a.length,p50:a[Math.ceil(a.length*.5)-1],p95:a[Math.ceil(a.length*.95)-1],max:a.at(-1)};};
const read=label=>JSON.parse(readFileSync(existsSync(prefix+label+'-performance.json')?prefix+label+'-performance.json':'.local/cube-c2b2-'+label+'-performance.json','utf8'));
const baseline=read('baseline'),final=read('final-auto'),forced=read('final-worker'),warmBase=read('warm-baseline'),warmFinal=read('warm-final');
assert.equal(baseline.trials.length,416);assert.equal(final.trials.length,416);assert.equal(forced.trials.length,64);assert.equal(warmBase.trials.length,144);assert.equal(warmFinal.trials.length,144);
const key=t=>[t.profile,t.n,t.tool,t.label,t.trial].join('/');
function match(a,b){const map=new Map(a.trials.map(t=>[key(t),t]));for(const t of b.trials){const ref=map.get(key(t));assert.ok(ref,key(t));for(const field of['sourceRevision','status','sourceSha256','finalSha256','historySha256','exactReplay'])assert.equal(t[field],ref[field],key(t)+' '+field);
 assert.deepEqual(t.commits.map(({ordinal,revision,center})=>({ordinal,revision,center})),ref.commits.map(({ordinal,revision,center})=>({ordinal,revision,center})),key(t)+' per-dab center/source');
 assert.deepEqual(t.operations,ref.operations,key(t)+' exact history');assert.equal(t.statistics.committedDabs,ref.statistics.committedDabs);assert.equal(t.statistics.writes,ref.statistics.writes);
 }return {pairs:b.trials.length,fullSourceAndFinalWordHashes:true,completeHistoryHashes:true,everyCommittedCenterAndSourceRevision:true,exactUndoRedo:true};}
const equivalence={default:match(baseline,final),forced:match(baseline,forced),warm:match(warmBase,warmFinal)};
function summary(trials){const pumps=trials.flatMap(t=>t.pumps),diag=trials.map(t=>t.diagnostics).filter(Boolean),stages=[...new Set(pumps.map(p=>p.stage))];
 return {trials:trials.length,complete:trials.filter(t=>t.status==='complete').length,failedPrefixes:trials.filter(t=>t.status==='failed').length,beginMs:q(trials.map(t=>t.beginMs)),acceptMs:q(trials.flatMap(t=>t.acceptMs)),firstCommitMs:q(trials.map(t=>t.firstCommitMs)),drainMs:q(trials.map(t=>t.drainMs)),dabsPerSecond:q(trials.map(t=>t.statistics.committedDabs/t.drainMs*1000)),mainPumpMs:q(pumps.map(p=>p.ms)),frameGapMs:q(trials.flatMap(t=>t.frameGaps)),longTasks:q(trials.flatMap(t=>t.longTasks)),stages:Object.fromEntries(stages.map(s=>[s,q(pumps.filter(p=>p.stage===s).map(p=>p.ms))])),
 routeCertificationMs:q(trials.flatMap(t=>t.stages.routeCertification??[])),spacingCenterMs:q(trials.flatMap(t=>t.stages.spacingCenter??[])),
 workerStages:Object.fromEntries([...new Set(diag.flatMap(d=>Object.keys(d.stages)))].map(s=>[s,{...q(diag.flatMap(d=>d.stages[s]?.samples??[])),lifetimeSamples:diag.reduce((n,d)=>n+(d.stages[s]?.count??0),0),totalMs:diag.reduce((n,d)=>n+(d.stages[s]?.totalMs??0),0),lifetimeMaxMs:Math.max(0,...diag.map(d=>d.stages[s]?.maxMs??0))}])),
 maxima:{pageHeapUsedBytes:Math.max(0,...trials.map(t=>t.memoryAfter.usedSize)),peakPendingInputs:Math.max(0,...trials.map(t=>t.statistics.path.peakPendingInputs)),peakStoredResponses:Math.max(0,...diag.map(d=>d.peakStoredResponses)),peakWorkers:Math.max(0,...diag.map(d=>d.peakWorkers)),sourceElevationCopies:Math.max(0,...diag.map(d=>d.sourceElevationCopies)),pathActionCopies:Math.max(0,...diag.map(d=>d.pathActionCopies)),replayActionBytes:Math.max(0,...diag.map(d=>d.replayActionBytes)),replayWorkUnits:Math.max(0,...diag.map(d=>d.replayWorkUnits)),comparisonBytes:Math.max(0,...diag.map(d=>d.comparisonBytes)),preparedDabBytes:Math.max(0,...trials.map(t=>t.statistics.peakPreparedDabBytes)),historyTypedBytes:Math.max(0,...trials.map(t=>t.historyStatistics.peakTypedBytes))},
 attachedTrials:diag.filter(d=>d.workerAttachments>0).length,fallbacks:diag.filter(d=>d.fallbackReason).map(d=>d.fallbackReason)};
}
const groups=[];for(const profile of['desktop','mobile'])for(const n of[128,256])for(const label of[...new Set(baseline.trials.map(t=>t.label))]){const filter=t=>t.profile===profile&&t.n===n&&t.label===label;groups.push({profile,n,label,baseline:summary(baseline.trials.filter(filter)),final:summary(final.trials.filter(filter))});}
const warmGroups=[];for(const profile of['desktop','mobile'])for(const n of[128,256])for(const label of['ordinary-edge','wide-edge','bending-65'])for(const temperature of['cold','warm']){const filter=t=>t.profile===profile&&t.n===n&&t.label===label&&(temperature==='cold'?t.trial===0:t.trial>0);warmGroups.push({profile,n,label,temperature,baseline:summary(warmBase.trials.filter(filter)),final:summary(warmFinal.trials.filter(filter))});}
const traces=[];for(const stage of['baseline','final'])for(const profile of['desktop','mobile']){const t=JSON.parse(readFileSync(prefix+stage+'-trace-'+profile+'-summary.json','utf8'));traces.push({stage,profile,trials:t.trialCount,tasks:t.tasks.length,fullTaskMs:q(t.tasks.map(t=>t.fullTaskMs)),executionOverlapMs:q(t.tasks.map(t=>t.executionOverlapMs))});}
const data={quantile:'Nearest rank, all retained samples; no warmup exclusions. Timing metadata keeps last 128 samples per stage and lifetime counts/maxima; workerStage quantiles use that explicitly bounded sample window. Main-pump/frame/raw browser data retain every sample. Cold and warm are reported separately.',equivalence,baseline:summary(baseline.trials),final:summary(final.trials),forced:summary(forced.trials),groups,warmGroups,traces,
 failedPrefixes:final.trials.filter(t=>t.status==='failed').map(t=>({key:key(t),committedDabs:t.statistics.committedDabs,reason:t.error,historyWords:t.operations[0]?.samples,exactReplay:t.exactReplay})),
 limitations:'Processing and frame/task scheduling only, no terrain pixels. Page-target CDP4x is not physical-phone evidence or equal worker CPU throttling. Whole tasks overlapping marks may include seeding/assertion time; clipped overlap excludes it. Page heap is not full worker RSS. Before/after collected sequentially, not randomized, with all actual samples retained.'};
writeFileSync(prefix+'performance-summary.json',JSON.stringify(data,null,2)+'\n');
const compressions=[];
for(const label of['final-auto','final-worker','warm-baseline','warm-final']){const path=prefix+label+'-performance.json',raw=readFileSync(path),gz=gzipSync(raw,{level:9}),expanded=gunzipSync(gz);assert.equal(sha(raw),sha(expanded));assert.equal(raw.length,expanded.length);writeFileSync(path+'.gz',gz);renameSync(path,'.local/cube-c2b2-'+label+'-performance.json');compressions.push({path:path+'.gz',rawBytes:raw.length,rawSha256:sha(raw),compressedBytes:gz.length,compressedSha256:sha(gz),exactDecompressedBytes:true});}
writeFileSync(prefix+'final-compression.json',JSON.stringify(compressions,null,2)+'\n');
console.log({equivalence,baselineComplete:data.baseline.complete,finalComplete:data.final.complete,failedPrefixes:data.final.failedPrefixes,forcedTrials:data.forced.trials,traces});
