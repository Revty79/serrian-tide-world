import assert from 'node:assert/strict';
import {execFileSync} from 'node:child_process';
import {readFileSync,writeFileSync,existsSync} from 'node:fs';
import {createHash} from 'node:crypto';
const baseline='bb0940dd2778f2277c1d9c7c5e4a498bfc48c272',implementation=execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim();assert.notEqual(implementation,baseline);
assert.equal(execFileSync('git',['rev-parse','HEAD^'],{encoding:'utf8'}).trim(),baseline);
const manifestPath='docs/evidence/cube-c2a-delivery-manifest.json',manifest=JSON.parse(readFileSync(manifestPath,'utf8'));
const paths=execFileSync('git',['diff','--name-only',baseline,implementation],{encoding:'utf8'}).trim().split(/\r?\n/).sort();assert.deepEqual(paths,[...manifest.entries.map(e=>e.path),manifestPath].sort());
const entries=paths.map(path=>{const row=execFileSync('git',['ls-tree',implementation,'--',path],{encoding:'utf8'}).trim(),head=row.split('\t')[0].split(' ');return{path,mode:head[0],gitBlob:head[2]};});
const manifestSha256=createHash('sha256').update(execFileSync('git',['show',`${implementation}:${manifestPath}`],{maxBuffer:20e6})).digest('hex');

writeFileSync(inventoryPath,JSON.stringify({baseline,implementation,files:entries.length,counts:manifest.counts,manifestPath,manifestSha256,entries},null,2)+'\n');
writeFileSync(receiptPath,`# Cube-C2A delivery receipt\n\nAccepted Cube-C1 baseline: \`${baseline}\`.\n\nImplementation: **\`${implementation}\`** ([GitHub commit](https://github.com/Revty79/serrian-tide-world/commit/${implementation})).\n\nThis receipt is added in the immediately following documentation commit, which is the delivered checkpoint on \`main\`.
