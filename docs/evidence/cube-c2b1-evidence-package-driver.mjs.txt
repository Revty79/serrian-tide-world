import {createHash} from 'node:crypto';
const prefix='docs/evidence/cube-c2b1-',directory='.local/cube-c2b1-browser/docs/evidence',sha=v=>createHash('sha256').update(v).digest('hex');
const entries=readdirSync(directory).sort().map(file=>{
 const path=`${directory}/${file}`,data=readFileSync(path);if(!statSync(path).isFile())throw Error('Expected flat evidence directory');
 const input=/^a7-p1c-saved-geography-n(128|256)\.json$/.test(file);if(input&&sha(data)!==sha(readFileSync(`docs/evidence/${file}`)))throw Error('Input fixture mismatch');

});
const archives=[];let current;

 current.uncompressedBytes+=entry.bytes;current.files++;entry.archive=current.archive;
}
