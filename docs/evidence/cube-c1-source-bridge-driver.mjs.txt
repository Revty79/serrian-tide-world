const sha=b=>createHash('sha256').update(b).digest('hex');

assert.equal(browser.files.length,256);assert.equal(final.files.length,256);assert.deepEqual(browser.files.map(f=>f.path),final.files.map(f=>f.path));
const changed=browser.files.flatMap((b,i)=>b.sha256!==final.files[i].sha256?[{path:b.path,browserSha256:b.sha256,finalSha256:final.files[i].sha256}]:[]);
assert.deepEqual(changed.map(f=>f.path).sort(),['src/terrain/solid-cube-brush.test.ts','src/terrain/solid-cube.ts']);
const snapshots={'src/terrain/solid-cube.ts':'docs/evidence/cube-c1-browser-terrain-source.ts.txt','src/terrain/solid-cube-brush.test.ts':'docs/evidence/cube-c1-browser-brush-test-source.ts.txt'};
for(const f of changed){assert.equal(sha(readFileSync(f.path)),f.finalSha256);assert.equal(sha(readFileSync(snapshots[f.path])),f.browserSha256);f.browserSourceSnapshot=snapshots[f.path];}

const runtimeSet=new Set(['src/terrain/solid-cube.ts','src/terrain/solid-cube-brush.ts','src/surfaces/cube-brush-distance.ts'].map(p=>resolve(p))),importers=[];

 const s=readFileSync(f.path,'utf8');for(const m of s.matchAll(/(?:from\s+|import\s*\(\s*)['"]([^'"]+)['"]/g))if(m[1].startsWith('.')){
  const target=resolve(dirname(f.path),m[1]);if(runtimeSet.has(target)){assert.ok(runtimeSet.has(resolve(f.path)),`${f.path} must not import the numerical-only cube terrain/dab modules`);importers.push({file:f.path,target:m[1]});}
 }
}
