import{spawn}from'node:child_process';import{createWriteStream,writeFileSync}from'node:fs';
const label=process.argv[2]??'before',at=Date.now(),log=createWriteStream(`docs/evidence/a7-p1c-${label}-measurement-log.txt`),child=spawn(process.execPath,['scripts/measure-sphere-standards.mjs',label,...process.argv.slice(3)],{windowsHide:true,stdio:['ignore','pipe','pipe']});child.stdout.pipe(log);child.stderr.pipe(log);const code=await new Promise(r=>child.on('exit',r));await new Promise(r=>log.end(r));const result={label,code,elapsedMs:Date.now()-at};writeFileSync(`docs/evidence/a7-p1c-${label}-measurement-result.json`,JSON.stringify(result,null,2));console.log(result);process.exitCode=code;
