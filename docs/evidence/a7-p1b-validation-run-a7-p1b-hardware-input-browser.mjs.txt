import { createWriteStream, readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { resolve } from 'node:path';

const server=spawn(process.execPath,['node_modules/vite/bin/vite.js','--host','127.0.0.1','--port','5223','--strictPort'],{windowsHide:true,stdio:['ignore','pipe','pipe']});
writeFileSync('.local/a7-p1b-hardware-input-server-pid.txt',String(server.pid));

while(Date.now()-started<30000){if(server.exitCode!==null)throw Error('Strict server failed to start');try{if((await fetch('http://127.0.0.1:5223')).ok){ready=true;break;}}catch{}await new Promise(r=>setTimeout(r,200));}

const child=spawn(process.execPath,['node_modules/@playwright/test/cli.js','test','--config','.local/a7-p1b-hardware-input-full-playwright.config.ts','--reporter=list,json'],{windowsHide:true,env:{...process.env,PLAYWRIGHT_JSON_OUTPUT_FILE:resolve('docs/evidence/a7-p1b-hardware-input-browser-results.json')},stdio:['ignore','pipe','pipe']});
child.stdout.on('data',b=>log.write(b));child.stderr.on('data',b=>log.write(b));
const code=await new Promise(r=>child.on('exit',r));

const after=snapshot(),changedPaths=[...new Set([...Object.keys(before),...Object.keys(after)])].filter(p=>before[p]!==after[p]);
writeFileSync('.local/a7-p1b-hardware-input-source-check.json',JSON.stringify({files:Object.keys(before).length,beforeDigest:sha(JSON.stringify(before)),afterDigest:sha(JSON.stringify(after)),unchanged:changedPaths.length===0,changedPaths},null,2));
let stopped=false;try{stopped=server.kill();}catch{}
